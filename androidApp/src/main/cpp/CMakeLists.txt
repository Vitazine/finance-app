cmake_minimum_required(VERSION 3.22.1)
project(finance_inference LANGUAGES C CXX)
if(NOT DEFINED LLAMA_SOURCE_DIR OR NOT EXISTS "${LLAMA_SOURCE_DIR}/include/llama.h")
  message(FATAL_ERROR "Provide the pinned llama.cpp checkout using -PllamaSourceDir")
endif()
execute_process(COMMAND git -C "${LLAMA_SOURCE_DIR}" rev-parse HEAD OUTPUT_VARIABLE LLAMA_REV OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT LLAMA_REV STREQUAL "060b2fea4ed528810ddc91d48fb22b3e4d63eb85")
  message(FATAL_ERROR "Unapproved llama.cpp revision")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
add_subdirectory("${LLAMA_SOURCE_DIR}" llama EXCLUDE_FROM_ALL)
add_library(finance_inference SHARED finance_inference.cpp)
target_compile_features(finance_inference PRIVATE cxx_std_17)
target_link_libraries(finance_inference PRIVATE llama)
